cmake_minimum_required(VERSION 3.16)
project(RdmaRangeLock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(rrl_rdma STATIC src/rdma/rdma_transport.cpp)
target_include_directories(rrl_rdma PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(rrl_rdma PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rrl_rdma PUBLIC ibverbs)
add_library(rrl_lock STATIC src/lock/lock_manager.cpp src/lock/remote_lock_manager.cpp)
target_include_directories(rrl_lock PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(rrl_lock PRIVATE -Wall -Wextra -Wpedantic)
add_library(rrl_index STATIC src/index/v1_btree_single.cpp)
target_include_directories(rrl_index PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(rrl_index PRIVATE -Wall -Wextra -Wpedantic)
add_library(rrl_index_latch STATIC src/index/v2_btree_latch.cpp)
target_include_directories(rrl_index_latch PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(rrl_index_latch PRIVATE -Wall -Wextra -Wpedantic)
add_library(rrl_txn STATIC src/txn/workload.cpp)
target_include_directories(rrl_txn PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(rrl_txn PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rrl_lock PUBLIC rrl_rdma)
target_link_libraries(rrl_txn PUBLIC rrl_index_latch rrl_lock)
add_executable(rrl_m1_range_lock_test test/m1_range_lock_test.cpp)

add_executable(rrl_v1_btree_test test/v1_btree_test.cpp)
target_link_libraries(rrl_v1_btree_test PRIVATE rrl_index)
target_compile_options(rrl_v1_btree_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_v2_btree_latch_test test/v2_btree_latch_test.cpp)
target_link_libraries(rrl_v2_btree_latch_test PRIVATE rrl_index_latch)
target_compile_options(rrl_v2_btree_latch_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_v3_phantom_test test/v3_phantom_test.cpp)
target_link_libraries(rrl_v3_phantom_test PRIVATE rrl_index_latch rrl_lock)
target_compile_options(rrl_v3_phantom_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_v4_gap_lock_test test/v4_gap_lock_test.cpp)
target_link_libraries(rrl_v4_gap_lock_test PRIVATE rrl_index_latch rrl_lock)
target_compile_options(rrl_v4_gap_lock_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_workload_trace_test test/workload_trace_test.cpp)
target_link_libraries(rrl_workload_trace_test PRIVATE rrl_txn)
target_compile_options(rrl_workload_trace_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_workload_runner benchmarks/workload_runner.cpp)
target_link_libraries(rrl_workload_runner PRIVATE rrl_txn rrl_index rrl_index_latch rrl_lock)
target_compile_options(rrl_workload_runner PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_rdma_lock_server benchmarks/rdma_lock_server.cpp)
target_link_libraries(rrl_rdma_lock_server PRIVATE rrl_lock rrl_rdma)
target_compile_options(rrl_rdma_lock_server PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rrl_rdma_lock_client test/rdma_lock_client.cpp)
target_link_libraries(rrl_rdma_lock_client PRIVATE rrl_lock rrl_rdma)
target_compile_options(rrl_rdma_lock_client PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()

add_test(NAME rrl_v3_phantom_test COMMAND rrl_v3_phantom_test)
